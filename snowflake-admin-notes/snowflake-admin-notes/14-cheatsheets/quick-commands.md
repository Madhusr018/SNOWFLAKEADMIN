# Quick Command Cheat Sheet

```sql
-- Context
USE ROLE x; USE WAREHOUSE x; USE DATABASE x; USE SCHEMA x;
-- Discover
SHOW DATABASES|SCHEMAS|TABLES|VIEWS|STAGES|PIPES|STREAMS|TASKS|WAREHOUSES|USERS|ROLES|GRANTS TO ROLE r|PARAMETERS IN ACCOUNT;
DESC TABLE t;  SELECT GET_DDL('TABLE','db.sch.t');
-- Admin
ALTER WAREHOUSE w SUSPEND|RESUME;   ALTER USER u SET DISABLED=TRUE;
SELECT SYSTEM$CANCEL_QUERY('qid');  SELECT SYSTEM$CANCEL_ALL_QUERIES(<session_id>);
-- Recovery
UNDROP TABLE t;   SELECT * FROM t AT(OFFSET=>-300);
-- Cost
SELECT * FROM SNOWFLAKE.ACCOUNT_USAGE.METERING_DAILY_HISTORY ORDER BY USAGE_DATE DESC LIMIT 30;
```
